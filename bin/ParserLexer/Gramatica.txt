//expresiones basicas
delim = \r|\n|\r\n
ESPACIO = {delim} | [ \t\f]
LETRA = [a-zA-Z]
DIGITO = [0-9]
ESCAPE = "\\"
DOBLECOMILLA = "\""
DIGITONOCERO = [1-9]
ID = "_"({LETRA}|{DIGITO})*"_"
FLOATNUM = -? (0 | {DIGITONOCERO} {DIGITO}*) ("." {DIGITO}+)? (("e" | "E") -? {DIGITO}+)?
NUM = {DIGITO}+ | {FLOATNUM}
STRCOMPLETO = {DOBLECOMILLA}({LETRA}|{DIGITO})({LETRA}|{DIGITO})*{DOBLECOMILLA}
STRPALABRA = {LETRA}({LETRA}|{DIGITO})*

//comentario de una linea
COMENTARIOSIMPLE = "#" [^\n]* {delim}?
//comentario multilinea
MULTICOMENTARIO = "\\_" ([^\\_] | "\\_")* "_/"
//var de cualquier tipo de comentario para registrarla
COMENTARIO = {COMENTARIOSIMPLE} | {MULTICOMENTARIO}

//palabras reservadas
 "rodolfo"         return INTEGER
  "bromista"       return FLOAT 
  "trueno"         return BOOL 
  "cupido"         return CHAR 
  "cometa"         return STRING 
  "false"          return FALSE 
  "true"           return TRUE 
  "abrecuento"     return BRACEIZQ 
  "cierracuento"   return BRACEDER 
  "abreempaque"    return CORCHETEIZQ 
  "cierraempaque"  return CORCHETEDER 
  "abreregalo"     return PARENIZQ 
  "cierraregalo"   return PARENDER 
  "finregalo"      return ENDEXPR 
  "entrega"        return ASIGNACION 
  "navidad"        return SUMA 
  "intercambio"    return RESTA 
  "reyes"          return DIVISION 
  "nochebuena"     return MULTIPLICACION 
  "magos"          return MODULO
  "adviento"       return POTENCIA 
  "quien"          return INCREMENTO 
  "grinch"         return DECREMENTO 
  "snowball"       return MENOR 
  "evergreen"      return MENORIGUAL 
  "minstix"        return MAYOR 
  "upatree"        return MAYORIGUAL 
  "mary"           return IGUALDAD 
  "openslae"       return DIFERENTE 
  "melchor"        return CONJUNCION 
  "gaspar"         return DISYUNCION 
  "baltazar"       return NEGACION 
  "elfo"           return IF 
  "hada"           return ELSE 
  "envuelve"       return WHILE 
  "duende"         return FOR 
  "varios"         return SWITCH 
  "historia"       return CASE 
  "último"         return DEFAULT 
  "corta"          return BREAK 
  "envia"          return RETURN 
  "sigue"          return DOSPUNTOS 
  "narra"          return PRINT
  "escucha"        return READ;
  "_verano_"       return MAIN;
  ","              return COMMA
  "'"              return COMILLA 
  "\""             return COMILLADOBLE 
  "!"              return CHAR//LO DEJAMOS COMO CHAR POR UN CODIGO DE EJEMPLO DEL PROFESOR
  STRCOMPLETO    return STRING
  STRPALABRA     return STRING
//numeros
  NUM         return NUMERO
//comentarios
  {COMENTARIO} { }
  "//".*            { }
  "/\\*"([^*]|\\*+[^*/])*"\\*/" { }
//ids
  ID    return IDENTIFICADOR
// Ignorar espacios en blanco
  {ESPACIO}    { }

//programa inicial
programa ::= _verano_ funciones

//definicion de funciones y sus producciones
funciones ::= funcion | funciones funcion
funcion ::= encabezadoFunc bloque
encabezadoFunc ::= tipos identificador abreregalo parametros cierraregalo
parametros ::= parametro | parametro "," parametros
parametro ::= tipos identificador
bloque ::= abrecuento sentencias cierracuento

//definicion de sentencias y sus producciones
sentencias ::= sentencia finregalo | sentencias sentencia finregalo
sentencia ::= creacionVariable 
                | creacionAsignacion 
                | expresion 
                | imprimir 
                | leer 
                | retornar 
                | controlFlujo

//declaraciones y asignaciones
creacionVariable  ::= tipos identificador entrega expresion
creacionAsignacion ::= identificador entrega expresion

//definicion de expresiones y sus producciones
expresion          ::= expresionRelacional 
                   | expresionAritmetica 
                   | llamadaFuncion
                   | literal

//literales
literal ::= digitos | letras | identificador

//llamadas
llamadaFuncion ::= identificador abreregalo argumentos cierraregalo
argumentos ::= expresion | expresion "," argumentos

//relacionales
expresionRelacional ::= expresion operadorRelacional expresion
operadorRelacional  ::= snowball | evergreen | minstix | upatree | mary | openslae

//aritmeticas
expresionAritmetica ::= expresion operadorAritmetico expresion
operadorAritmetico  ::= navidad | intercambio | reyes | nochebuena | magos | adviento

//definicion de estrucutras de control
controlFlujo ::= if | while | for | switch
if  ::= elfo abreregalo expresionRelacional cierraregalo bloque (hada bloque)?
while ::= envuelve abreregalo expresionRelacional cierraregalo bloque
for ::= duende abreregalo inicializacion ";" expresionRelacional ";" incremento cierraregalo bloque
incremento  ::= identificador quien | identificador grinch
inicializacion  ::= creacionVariable | creacionAsignacion
switch ::= varios abreregalo identificador cierraregalo abrecuento casos cierracuento
casos  ::= caso | casos caso
caso   ::= historia literal sigue bloque | último sigue bloque

//retronos
retornar ::= envia expresion finregalo
imprimir ::= narra expresion finregalo
leer ::= escucha identificador finregalo

//tipos y tokens
tipos ::= rodolfo | bromista | trueno | cupido | cometa
identificador = letra (letra | digito | "_")*
digito = [0-9]
letra = [a-zA-Z]
abreregalo = "("
cierraregalo = ")"
abrecuento = "{"
cierracuento = "}"
entrega = "="
finregalo = ";"
